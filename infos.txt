@router.get("/communes", response_model=list[CommuneOut])
async def list_communes(
    wilaya_id: int | None = None,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> list[CommuneOut]:
    """Retourne les communes, filtrables par wilaya_id. Liste vide si aucune commune n'existe."""
    stmt = select(Commune).order_by(Commune.wilaya_id, Commune.name)
    if wilaya_id is not None:
        stmt = stmt.where(Commune.wilaya_id == wilaya_id)
    result = await db.execute(stmt)
    return result.scalars().all()

"""
ModÃ¨les ORM â€” Utilisateurs & RÃ´les.
Correspond exactement aux tables `users` et `roles` de
database/migrations/001_init_core_schema.sql (Phase 2). Ne pas dupliquer
la dÃ©finition du schÃ©ma ailleurs : cette migration reste la source de vÃ©ritÃ©
pour les contraintes et index ; ce fichier ne fait que la mapper pour l'ORM.
"""
import uuid
from datetime import datetime, timezone

from sqlalchemy import Boolean, ForeignKey, SmallInteger, String, DateTime, JSON, func
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.db.session import Base


class Role(Base):
    __tablename__ = "roles"

    id: Mapped[int] = mapped_column(SmallInteger, primary_key=True)
    code: Mapped[str] = mapped_column(String, unique=True, nullable=False)
    label: Mapped[str] = mapped_column(String, nullable=False)
    permissions: Mapped[dict] = mapped_column(JSON, nullable=False, default=dict)

    users: Mapped[list["User"]] = relationship(back_populates="role")


def _now() -> datetime:
    return datetime.now(timezone.utc)


class User(Base):
    __tablename__ = "users"

    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True), primary_key=True, default=uuid.uuid4
    )
    email: Mapped[str] = mapped_column(String, unique=True, nullable=False)
    password_hash: Mapped[str] = mapped_column(String, nullable=False)
    first_name: Mapped[str | None] = mapped_column(String, nullable=True)
    last_name: Mapped[str | None] = mapped_column(String, nullable=True)
    role_id: Mapped[int] = mapped_column(ForeignKey("roles.id"), nullable=False)
    is_active: Mapped[bool] = mapped_column(Boolean, nullable=False, default=True)
    last_login_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=_now)
    updated_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=_now, onupdate=_now)
    deleted_at: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))

    role: Mapped["Role"] = relationship(back_populates="users")

    @property
    def full_name(self) -> str:
        return " ".join(part for part in [self.first_name, self.last_name] if part) or self.email
